    <script>
        // Simulated wallet data
        let walletData = null;

        function addWallet() {
            const address = document.getElementById('walletAddress').value.trim();
            
            if (!address) {
                alert('Please enter a wallet address');
                return;
            }

            // Simulate fetching wallet data (in real app, this would call an API)
            walletData = {
                address: address,
                totalValue: 17284.73,
                change24h: 3.85,
                changeValue: 645.50,
                tokens: [
                    { symbol: 'ETH', name: 'Ethereum', amount: 2.45, price: 4125.00, change: 5.2, value: 10106.25 },
                    { symbol: 'SOL', name: 'Solana', amount: 45.2, price: 150.00, change: -1.5, value: 6780.00 },
                    { symbol: 'USDC', name: 'USD Coin', amount: 2100.00, price: 1.00, change: 0.0, value: 2100.00 },
                    { symbol: 'PEPE', name: 'Pepe', amount: 50000000, price: 0.000012, change: 12.5, value: 600.00 },
                    { symbol: 'WIF', name: 'dogwifhat', amount: 125.5, price: 2.45, change: -3.2, value: 307.48 }
                ]
            };

            displayHoldings();
            
            // Update total portfolio value
            document.getElementById('totalValue').textContent = '$' + walletData.totalValue.toLocaleString('en-US', {minimumFractionDigits: 2, maximumFractionDigits: 2});
            const changeClass = walletData.change24h >= 0 ? 'positive' : 'negative';
            const sign = walletData.change24h >= 0 ? '+' : '';
            document.getElementById('portfolioChange').innerHTML = 
                `<span class="${changeClass}">${sign}${walletData.change24h.toFixed(2)}% (+$${walletData.changeValue.toLocaleString()})</span>`;
        }

        function displayHoldings() {
            const section = document.getElementById('holdingsSection');
            
            if (!walletData) {
                section.innerHTML = '<div class="empty-state"><p>Add your wallet address above to see your holdings and P/L</p></div>';
                return;
            }

            let html = '';
            walletData.tokens.forEach(token => {
                const changeClass = token.change >= 0 ? 'positive' : 'negative';
                const sign = token.change >= 0 ? '+' : '';
                
                html += `
                    <div class="holding-item">
                        <div class="holding-info">
                            <div class="token-icon">${token.symbol}</div>
                            <div>
                                <div style="font-weight:600">${token.name}</div>
                                <div style="color:#8892b0; font-size:0.9rem">${token.amount} ${token.symbol}</div>
                            </div>
                        </div>
                        <div style="text-align:right">
                            <div style="font-weight:600">$${token.value.toLocaleString('en-US', {minimumFractionDigits: 2, maximumFractionDigits: 2})}</div>
                            <div class="${changeClass}">${sign}${token.change}%</div>
                        </div>
                    </div>
                `;
            });

            section.innerHTML = html;
        }

        // Simulate live price ticking
        setInterval(() => {
            if (walletData) {
                walletData.totalValue += (Math.random() - 0.5) * 50;
                document.getElementById('totalValue').textContent = 
                    '$' + walletData.totalValue.toLocaleString('en-US', {minimumFractionDigits: 2, maximumFractionDigits: 2});
            }
        }, 3000);
    </script>
